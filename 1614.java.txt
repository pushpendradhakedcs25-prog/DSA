class Solution {
    public long findKthSmallest(int[] coins, int k) {
        HashSet<Integer> set = new HashSet<>();
      
        for(int i=0; i<coins.length; i++){
            long c=coins[i];
            int mul=1;
            while(mul<=(long)k*c){
            set.add(mul);
            mul++;}
        }
      List<Long> numbers = new ArrayList<>(set);
        Collections.sort(numbers);
        return number.get(k+1);
    }
}
